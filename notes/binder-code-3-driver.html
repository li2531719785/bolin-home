第 4 章 服务端

第 2 章的 talkWithDriver 最后把 binder_write_read 交给 ioctl；本章从那次 ioctl 进入驱动，只跟一条 BC_TRANSACTION：用户空间把命令写进来，驱动找到目标、切缓冲区、拷贝数据、入队，目标进程再从 binder_thread_read 读走 BR_TRANSACTION。

下面五个函数来自 drivers/android/binder.c；中场会认 binder_proc、binder_node、binder_ref、binder_thread、binder_transaction 和 binder_buffer。

为方便阅读，代码块里的 // ← 是本站添加的阅读标注；// ……（省略：…） 是节选标记，被省略的是与主线无关的噪声段或旁支。代码语句本身保持原样。

一、binder_ioctl

先把 ioctl 当成分发器：它只认命令号，把 BINDER_WRITE_READ 交给读写函数。

static long binder_ioctl(struct file *filp, unsigned int cmd, unsigned long arg)
{
	int ret;
	struct binder_proc *proc = filp->private_data;
	struct binder_thread *thread;
	void __user *ubuf = (void __user *)arg;

	/*pr_info("binder_ioctl: %d:%d %x %lx\n",
			proc->pid, current->pid, cmd, arg);*/

	binder_selftest_alloc(&proc->alloc);

	trace_binder_ioctl(cmd, arg);

	// ……（省略：BINDER_THREAD_EXIT case）

binder_ioctl（5673 行）先通过 binder_get_thread 找到当前线程的 binder_thread，再在 switch 里分发。BINDER_WRITE_READ 分支交给 binder_ioctl_write_read；设置线程数、context manager、版本等其它 case 与本章主线无关。

写和读都在 binder_ioctl_write_read：下一节看它怎么把一次 ioctl 拆成先写后读。

二、binder_ioctl_write_read

一次 ioctl 同时带着写缓冲和读缓冲：先把用户空间的 binder_write_read 拿进来，再按 write_size / read_size 分别处理。

static int binder_ioctl_write_read(struct file *filp, unsigned long arg,
				struct binder_thread *thread)
{
	int ret = 0;
	struct binder_proc *proc = filp->private_data;
	void __user *ubuf = (void __user *)arg;
	struct binder_write_read bwr;
	bool has_special_work = false;

	if (copy_from_user(&bwr, ubuf, sizeof(bwr))) { // ← 把用户空间的 binder_write_read 拿进来
		ret = -EFAULT;
		goto out;
	}
	binder_debug(BINDER_DEBUG_READ_WRITE,
		     "%d:%d write %lld at %016llx, read %lld at %016llx\n",
		     proc->pid, thread->pid,
		     (u64)bwr.write_size, (u64)bwr.write_buffer,
		     (u64)bwr.read_size, (u64)bwr.read_buffer);

	if (bwr.write_size > 0) { // ← 先处理写
		ret = binder_thread_write(proc, thread,
					  bwr.write_buffer,
					  bwr.write_size,
					  &bwr.write_consumed);
		trace_binder_write_done(ret);
		if (ret < 0) {
			bwr.read_consumed = 0;
			if (copy_to_user(ubuf, &bwr, sizeof(bwr)))
				ret = -EFAULT;
			goto out;
		}
	}
	if (bwr.read_size > 0) {
		ret = binder_thread_read(proc, thread, bwr.read_buffer, // ← 再处理读
					 bwr.read_size,
					 &bwr.read_consumed,
					 filp->f_flags & O_NONBLOCK);
		trace_binder_read_done(ret);
		binder_inner_proc_lock(proc);
		trace_android_vh_binder_has_proc_work_ilocked(
			thread, true, &has_special_work);
		if (!binder_worklist_empty_ilocked(&proc->todo) || has_special_work)
			binder_wakeup_proc_ilocked(proc);
		binder_inner_proc_unlock(proc);
		trace_android_vh_binder_read_done(proc, thread);
		if (ret < 0) {
			if (copy_to_user(ubuf, &bwr, sizeof(bwr)))
				ret = -EFAULT;
			goto out;
		}
	}
	binder_debug(BINDER_DEBUG_READ_WRITE,
		     "%d:%d wrote %lld of %lld, read return %lld of %lld\n",
		     proc->pid, thread->pid,
		     (u64)bwr.write_consumed, (u64)bwr.write_size,
		     (u64)bwr.read_consumed, (u64)bwr.read_size);
	if (copy_to_user(ubuf, &bwr, sizeof(bwr))) { // ← 回写 consumed，第 2 章靠它清 mOut
		ret = -EFAULT;
		goto out;
	}
out:
	return ret;
}

binder_ioctl_write_read（5390–5452 行）先 copy_from_user 取第 2 章 talkWithDriver 填的 binder_write_read；write_size > 0 时调用 binder_thread_write，read_size > 0 时调用 binder_thread_read；最后把 write_consumed / read_consumed 写回用户空间，第 2 章靠它清空 mOut、设置 mIn。

先看写侧：binder_thread_write 怎么把 BC_TRANSACTION 从用户缓冲里取出来？

三、binder_thread_write

写侧逐个读 BC_* 命令；主线只跟 BC_TRANSACTION / BC_REPLY，其它命令折叠。

static int binder_thread_write(struct binder_proc *proc,
			struct binder_thread *thread,
			binder_uintptr_t binder_buffer, size_t size,
			binder_size_t *consumed)
{
	uint32_t cmd;
	struct binder_context *context = proc->context;
	void __user *buffer = (void __user *)(uintptr_t)binder_buffer;
	void __user *ptr = buffer + *consumed;
	void __user *end = buffer + size;

	while (ptr < end && thread->return_error.cmd == BR_OK) {
		int ret;

		if (get_user(cmd, (uint32_t __user *)ptr))
			return -EFAULT;
		ptr += sizeof(uint32_t);
		trace_binder_command(cmd);
		if (_IOC_NR(cmd) < ARRAY_SIZE(binder_stats.bc)) {
			atomic_inc(&binder_stats.bc[_IOC_NR(cmd)]);
			atomic_inc(&proc->stats.bc[_IOC_NR(cmd)]);
			atomic_inc(&thread->stats.bc[_IOC_NR(cmd)]);
		}
		switch (cmd) {
		// ……（省略：BC_INCREFS / BC_ACQUIRE / BC_RELEASE / BC_DECREFS 等引用计数命令）
		// ……（省略：BC_INCREFS_DONE / BC_ACQUIRE_DONE / BC_ATTEMPT_ACQUIRE / BC_ACQUIRE_RESULT）
		// ……（省略：BC_REGISTER_LOOPER / BC_ENTER_LOOPER / BC_EXIT_LOOPER）
		// ……（省略：BC_REQUEST_DEATH_NOTIFICATION / BC_CLEAR_DEATH_NOTIFICATION / BC_DEAD_BINDER_DONE）
		// ……（省略：BC_TRANSACTION_SG / BC_REPLY_SG）
		case BC_FREE_BUFFER: {
			binder_uintptr_t data_ptr;
			struct binder_buffer *buffer;

			if (get_user(data_ptr, (binder_uintptr_t __user *)ptr))
				return -EFAULT;
			ptr += sizeof(binder_uintptr_t);
			// ……（省略：BC_FREE_BUFFER 查找失败时的报错）
			buffer = binder_alloc_prepare_to_free(&proc->alloc,
							      data_ptr);
			binder_free_buf(proc, thread, buffer, false);
			break;
		}
		case BC_TRANSACTION:
		case BC_REPLY: {
			struct binder_transaction_data tr;

			if (copy_from_user(&tr, ptr, sizeof(tr))) // ← 把事务数据从用户空间拿进来
				return -EFAULT;
			ptr += sizeof(tr);
			binder_transaction(proc, thread, &tr,
					   cmd == BC_REPLY, 0); // ← 真正干活的地方
			break;
		}
		// ……（省略：BC_TRANSACTION_SG / looper / 死亡通知等命令）
		default:
			pr_err("%d:%d unknown command %u\n",
			       proc->pid, thread->pid, cmd);
			return -EINVAL;
		}
		*consumed = ptr - buffer;
	}
	return 0;
}

binder_thread_write（4121 行）从用户空间循环取命令号；BC_TRANSACTION / BC_REPLY 分支把 binder_transaction_data 拷进来（tr 正是 第 2 章 writeTransactionData 填的 binder_transaction_data），再调用 binder_transaction。BC_FREE_BUFFER 分支回收第 2 章 freeBuffer 归还的缓冲区。

BC_TRANSACTION 的全部工作都在 binder_transaction：下一节读这个最长函数的主线。

四、binder_transaction

这里只走一次非 reply、非 oneway 的主线：找目标、切缓冲区、拷贝数据、入队唤醒；reply 分支和错误标签群折叠。

binder_transaction（3184 行）的主线可以拆成四步：找目标——handle 查 ref，再从 ref->node 拿到 target_proc；切 buffer——在目标进程的 alloc 里分配；拷贝——binder_alloc_copy_user_to_buffer 把偏移表和数据搬过去；入队——同步调用先压 transaction_stack，再由 binder_proc_transaction（3024 行）选择目标线程并唤醒。

fixup 循环遍历偏移表，把 BINDER_TYPE_BINDER / HANDLE 等对象翻译成目标进程认识的引用；binder_translate_handle 只是这个循环里的一个分支。目标进程的 binder_alloc 就是它启动时 mmap 的区域，所以第 2 章 ipcSetDataReference 能直接接地址。

六、binder_thread_read

读侧是写侧的镜像：从 todo 取 work，把事务或完成通知转换成 BR_* 写回用户缓冲。

static int binder_thread_read(struct binder_proc *proc,
			      struct binder_thread *thread,
			      binder_uintptr_t binder_buffer, size_t size,
			      binder_size_t *consumed, int non_block)
{
	void __user *buffer = (void __user *)(uintptr_t)binder_buffer;
	void __user *ptr = buffer + *consumed;
	void __user *end = buffer + size;

	int ret = 0;
	int wait_for_proc_work;

	if (*consumed == 0) {
		if (put_user(BR_NOOP, (uint32_t __user *)ptr)) // ← 占位符，保证 consumed 不为 0
			return -EFAULT;
		ptr += sizeof(uint32_t);
	}

retry:
	binder_inner_proc_lock(proc);
	wait_for_proc_work = binder_available_for_proc_work_ilocked(thread);
	binder_inner_proc_unlock(proc);

	thread->looper |= BINDER_LOOPER_STATE_WAITING;
	if (wait_for_proc_work) {
		// ……（省略：looper 状态检查与优先级恢复）
	}

	if (non_block) {
		// ……（省略：非阻塞 -EAGAIN 细节）
	} else {
		ret = binder_wait_for_work(thread, wait_for_proc_work); // ← 没活干就睡，直到 todo 有工作
	}

	thread->looper &= ~BINDER_LOOPER_STATE_WAITING;
	if (ret)
		return ret;

	while (1) {
		uint32_t cmd;
		struct binder_transaction_data_secctx tr;
		struct binder_transaction_data *trd = &tr.transaction_data;
		struct binder_work *w = NULL;
		struct list_head *list = NULL;
		struct binder_transaction *t = NULL;

		binder_inner_proc_lock(proc);
		// ……（省略：特殊 worklist / nothing_to_do 检查）
		if (!binder_worklist_empty_ilocked(&thread->todo))
			list = &thread->todo;
		else if (!binder_worklist_empty_ilocked(&proc->todo) &&
			   wait_for_proc_work)
			list = &proc->todo;
		else {
			// ……（省略：无工作时的 retry/break）
			break;
		}

		w = binder_dequeue_work_head_ilocked(list); // ← 从 todo 取一条工作
		switch (w->type) {
		case BINDER_WORK_TRANSACTION: {
			binder_inner_proc_unlock(proc);
			t = container_of(w, struct binder_transaction, work);
		} break;
		case BINDER_WORK_TRANSACTION_COMPLETE: {
			cmd = BR_TRANSACTION_COMPLETE;
			binder_inner_proc_unlock(proc);
			kfree(w);
			if (put_user(cmd, (uint32_t __user *)ptr))
				return -EFAULT;
			ptr += sizeof(uint32_t);
		} break;
		// ……（省略：RETURN_ERROR / NODE / DEAD_BINDER 等 case）
		}
		if (!t)
			continue;

		BUG_ON(t->buffer == NULL);
		if (t->buffer->target_node) {
			struct binder_node *target_node = t->buffer->target_node;

			trd->target.ptr = target_node->ptr;
			trd->cookie = target_node->cookie;
			cmd = BR_TRANSACTION; // ← 有 target_node 是服务端新请求
		} else {
			trd->target.ptr = 0;
			trd->cookie = 0;
			cmd = BR_REPLY; // ← 否则是回程，第 2 章 waitForResponse 等的那条
		}
		trd->code = t->code;
		trd->flags = t->flags;
		// ……（省略：sender_pid / fd fixups / 安全上下文）
		trd->data_size = t->buffer->data_size;
		trd->offsets_size = t->buffer->offsets_size;
		trd->data.ptr.buffer = t->buffer->user_data; // ← 把 mmap 区地址告诉用户空间
		trd->data.ptr.offsets = trd->data.ptr.buffer +
					ALIGN(t->buffer->data_size,
					    sizeof(void *));
		if (put_user(cmd, (uint32_t __user *)ptr))
			return -EFAULT;
		ptr += sizeof(uint32_t);
		// ……（省略：copy_to_user 事务数据、事务栈处理）
	}
	*consumed = ptr - buffer;
	return 0;
}

binder_thread_read（4697 行）先用 BR_NOOP（4712 行附近）占住读缓冲；没有工作时 binder_wait_for_work（4743 行附近）让线程睡眠。醒来后从线程或进程的 todo 取 work（4789 行附近）：BR_TRANSACTION_COMPLETE 让同步调用继续等，BINDER_WORK_TRANSACTION 则按 target_node 分成服务端新请求 BR_TRANSACTION 和回程 BR_REPLY（4973 行附近；也就是第 2 章 waitForResponse 等的那条）。

trd->data.ptr.buffer = t->buffer->user_data（5031 行附近）把目标进程 mmap 区的地址写回用户空间；第 2 章 ipcSetDataReference 接的就是这个地址。

BR_TRANSACTION 是第 4 章服务端的入口；BR_REPLY 已在第 2 章见过——本章至此闭环。
